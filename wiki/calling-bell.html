Calling Bell & Talking Clock — Start With One Unit

One-unit calling bell and talking clock. 220 V AC, strictly local-only. ESP-12F + PCF8563 + PAM8403 + 433 MHz EV1527. Sample ₹1,738; production target ₹540 at five units.

Bird's Eye View

Outdoor button → chime. On the hour → spoken time, muted during bedtime. Local Wi-Fi browser syncs clock and uploads chimes without cloud or router time servers. CR2032 maintains time through power cuts.

Signal flow: button → 433 MHz receiver → ESP-12F → 1-bit Delta-Sigma audio on GPIO3 → RC filter → PAM8403 → speaker. Isolated SMPS drops 220 V AC to 5 V; AMS1117 provides 3.3 V for logic and radio.

Physical Breadboard Layout (Step 3 Test Unit)

Breadboard for Step 3; Step 8 moves to production PCB. Rails: red +5 V, orange +3.3 V, blue GND.

   RED RAIL (+5V)     SMPS +5V ── AMS1117 IN ── PAM8403 VCC
   ORANGE RAIL (+3.3V) AMS1117 OUT ── ESP-12F VCC ── PCF8563 VCC ── RXB12 VCC
   BLUE RAIL (GND)    every module's GND joined at ONE star point

   ┌──────────────────────── BREADBOARD (top to bottom) ──────────────────────┐
   │ A  HLK-5M05 SMPS board   : 220 V AC in, 5 V out. MAINS - ELECTRICIAN.   │
   │ B  AMS1117-3.3 + caps    : 5 V to 3.3 V rail for logic and receiver.    │
   │ C  ESP-12F on adapter    : 5-pin safe flash header; 10k strapping;      │
   │                            GPIO3 (audio), GPIO4/5 (I2C), GPIO14 (RF).   │
   │ D  PCF8563 + crystal     : CR2032 coin cell + 2 diodes beside it.       │
   │ E  RXB12/RXB6 + antenna  : VCC from 3.3 V, 100nF + 10µF right at pins.   │
   │ F  1kΩ + 0.1µF, PAM8403  : GPIO3 -> RC filter -> PAM8403 L_in -> spk.  │
   └──────────────────────────────────────────────────────────────────────────┘
   CP2102 3.3V pin powers the board during flashing; mains remains unplugged.

Two Basic Functions & Data Path

Both functions stream 8 kHz / 8-bit unsigned mono PCM from LittleFS to GPIO3 Delta-Sigma DMA → PAM8403 → speaker.

  1. Calling Bell: EV1527 24-bit frame arrives on GPIO14. ISR decodes code; when matched against paired address, firmware streams /audio/chime.raw (~40 kB). Unplug speaker wire during UART flashing to silence RX data chatter.
  2. Hourly Talking Clock: Every minute firmware queries PCF8563 via I2C. At HH:00 outside quiet hours (e.g. 22:00–06:00), it stitches vocabulary clips from LittleFS (~160 kB total: intro.raw + hour + oclock.raw + am/pm.raw). Inside quiet hours it stays silent.

Idle Hiss Reality: Prototype vs Production

On the breadboard prototype, the ₹28 PAM8403 breakout module hardwires /SHDN to 5 V, so faint background hiss is normal during quiet standby. In Step 8 production PCBA, GPIO12 routes directly to PAM8403 SOP-16 pin 1 (/SHDN), pulling it LOW when idle to eliminate 100% of idle hiss.

Step 5 — Core Firmware MVP (Data Pipelines & Control Engine)

Complete C++ firmware for Arduino ESP8266 core with LittleFS storage, non-blocking EV1527 ISR decoding, PCF8563 BCD timekeeping, 1-bit Delta-Sigma DMA audio streaming, and REST API.

Calling Bell & Talking Clock Firmware (firmware.ino) Production Ready
Target: ESP-12F (4MB Flash, 3MB LittleFS) Libraries: LittleFS, Wire, ESP8266WiFi, ESP8266WebServer, ArduinoJson 
#include <ESP8266WiFi.h>
#include <ESP8266WebServer.h>
#include <LittleFS.h>
#include <Wire.h>
#include <ArduinoJson.h>
#include <i2s.h>

#define PIN_AUDIO_DATA 3   // I2S0_DATA (GPIO3)
#define PIN_AMP_SHDN   12  // PAM8403 /SHDN (Step 8 PCB; bench module ties to VCC)
#define PIN_RF_DATA    14  // 433 MHz RXB12 DATA (Interrupt)
#define PIN_I2C_SDA    4   // PCF8563 SDA
#define PIN_I2C_SCL    5   // PCF8563 SCL
#define PCF8563_ADDR   0x51
#define CONFIG_PATH    "/config.json"

struct BellConfig {
  uint32_t paired_rf_id = 0x5A3C1E;
  char chime_file[32] = "/audio/chime.raw";
  bool clock_enabled = true;
  uint8_t quiet_start = 22;
  uint8_t quiet_end = 6;
} config;

ESP8266WebServer server(80);
volatile bool g_pairing_mode = false;
volatile uint32_t g_last_rf_code = 0;
volatile bool g_rf_packet_ready = false;
volatile uint32_t g_last_edge_us = 0;
volatile uint32_t g_rf_shift_reg = 0;
volatile uint8_t g_rf_bit_count = 0;

// --- EV1527 24-Bit OOK RF Interrupt Decoder ---
void ICACHE_RAM_ATTR isr_rf_edge() {
  uint32_t now = micros();
  uint32_t duration = now - g_last_edge_us;
  g_last_edge_us = now;
  bool level = digitalRead(PIN_RF_DATA);

  if (!level) { // Measured HIGH pulse duration
    if (duration > 7500 && duration < 12500) {
      g_rf_bit_count = 0;
      g_rf_shift_reg = 0;
    } else if (duration > 180 && duration < 500) {
      g_rf_shift_reg = (g_rf_shift_reg << 1);
      g_rf_bit_count++;
    } else if (duration > 700 && duration < 1200) {
      g_rf_shift_reg = (g_rf_shift_reg << 1) | 1;
      g_rf_bit_count++;
    } else {
      g_rf_bit_count = 0;
    }
    if (g_rf_bit_count == 24) {
      g_last_rf_code = g_rf_shift_reg;
      g_rf_packet_ready = true;
      g_rf_bit_count = 0;
    }
  }
}

// --- PCF8563 RTC Driver ---
uint8_t dec2bcd(uint8_t v) { return ((v / 10) << 4) | (v % 10); }
uint8_t bcd2dec(uint8_t v) { return ((v >> 4) * 10) + (v & 0x0F); }

struct RTCDateTime {
  uint16_t year; uint8_t month; uint8_t day;
  uint8_t hour; uint8_t minute; uint8_t second;
  bool valid;
};

RTCDateTime rtc_read_time() {
  RTCDateTime dt = {0,0,0,0,0,0,false};
  Wire.beginTransmission(PCF8563_ADDR);
  Wire.write(0x02);
  if (Wire.endTransmission() != 0) return dt;
  Wire.requestFrom(PCF8563_ADDR, 7);
  if (Wire.available() < 7) return dt;

  uint8_t raw_sec = Wire.read();
  dt.valid = !(raw_sec & 0x80); // VL bit: 0=valid, 1=power-cut corrupted
  dt.second = bcd2dec(raw_sec & 0x7F);
  dt.minute = bcd2dec(Wire.read() & 0x7F);
  dt.hour = bcd2dec(Wire.read() & 0x3F);
  dt.day = bcd2dec(Wire.read() & 0x3F);
  Wire.read(); // weekday
  dt.month = bcd2dec(Wire.read() & 0x1F);
  dt.year = 2000 + bcd2dec(Wire.read());
  return dt;
}

void rtc_write_time(const RTCDateTime &dt) {
  Wire.beginTransmission(PCF8563_ADDR);
  Wire.write(0x02);
  Wire.write(dec2bcd(dt.second) & 0x7F);
  Wire.write(dec2bcd(dt.minute));
  Wire.write(dec2bcd(dt.hour));
  Wire.write(dec2bcd(dt.day));
  Wire.write(0x00);
  Wire.write(dec2bcd(dt.month));
  Wire.write(dec2bcd(dt.year % 100));
  Wire.endTransmission();
}

// --- 1-Bit Delta-Sigma DMA Audio Engine ---
static inline uint32_t sigma_delta_32(uint8_t sample, int16_t &acc) {
  uint32_t out_word = 0;
  int16_t in_val = (int16_t)sample - 128;
  for (int b = 0; b < 32; b++) {
    acc += in_val;
    if (acc >= 0) { acc -= 127; out_word |= (1UL << b); }
    else { acc += 128; }
  }
  return out_word;
}

void play_audio_file(const char *path) {
  if (!LittleFS.exists(path)) return;
  File f = LittleFS.open(path, "r");
  if (!f) return;

  digitalWrite(PIN_AMP_SHDN, HIGH); // Unmute amplifier
  delay(10);

  i2s_begin();
  i2s_set_rate(8000);

  uint8_t chunk[256];
  int16_t acc = 0;
  while (f.available()) {
    size_t n = f.read(chunk, sizeof(chunk));
    for (size_t i = 0; i < n; i++) {
      i2s_write_sample(sigma_delta_32(chunk[i], acc));
    }
    yield();
  }
  f.close();
  i2s_end();
  digitalWrite(PIN_AMP_SHDN, LOW); // Mute on production board
}

void speak_time(uint8_t hour_24) {
  uint8_t hour_12 = hour_24 % 12;
  if (hour_12 == 0) hour_12 = 12;
  bool is_pm = (hour_24 >= 12);

  play_audio_file("/audio/intro.raw");
  char hpath[32];
  snprintf(hpath, sizeof(hpath), "/audio/%02d.raw", hour_12);
  play_audio_file(hpath);
  play_audio_file("/audio/oclock.raw");
  play_audio_file(is_pm ? "/audio/pm.raw" : "/audio/am.raw");
}

bool in_quiet_window(uint8_t hr, uint8_t start, uint8_t end) {
  if (start > end) return (hr >= start || hr < end);
  return (hr >= start && hr < end);
}

void load_configuration() {
  if (!LittleFS.exists(CONFIG_PATH)) return;
  File f = LittleFS.open(CONFIG_PATH, "r");
  StaticJsonDocument<256> doc;
  if (deserializeJson(doc, f) == DeserializationError::Ok) {
    config.paired_rf_id = doc["rf_id"] | config.paired_rf_id;
    strlcpy(config.chime_file, doc["chime"] | config.chime_file, sizeof(config.chime_file));
    config.clock_enabled = doc["clock"] | config.clock_enabled;
    config.quiet_start = doc["q_start"] | config.quiet_start;
    config.quiet_end = doc["q_end"] | config.quiet_end;
  }
  f.close();
}

void save_configuration() {
  File f = LittleFS.open(CONFIG_PATH, "w");
  StaticJsonDocument<256> doc;
  doc["rf_id"] = config.paired_rf_id;
  doc["chime"] = config.chime_file;
  doc["clock"] = config.clock_enabled;
  doc["q_start"] = config.quiet_start;
  doc["q_end"] = config.quiet_end;
  serializeJson(doc, f);
  f.close();
}

void setup_http_routes() {
  server.on("/api/status", HTTP_GET, []() {
    RTCDateTime dt = rtc_read_time();
    StaticJsonDocument<256> res;
    char tbuf[32];
    snprintf(tbuf, sizeof(tbuf), "%04d-%02d-%02d %02d:%02d:%02d",
             dt.year, dt.month, dt.day, dt.hour, dt.minute, dt.second);
    res["time"] = tbuf;
    res["rtc_valid"] = dt.valid;
    res["paired_rf_id"] = config.paired_rf_id;
    res["quiet_active"] = in_quiet_window(dt.hour, config.quiet_start, config.quiet_end);
    FSInfo fs; LittleFS.info(fs);
    res["flash_free_bytes"] = fs.totalBytes - fs.usedBytes;
    String out; serializeJson(res, out);
    server.send(200, "application/json", out);
  });

  server.on("/api/time", HTTP_POST, []() {
    if (!server.hasArg("plain")) { server.send(400, "text/plain", "Missing body"); return; }
    StaticJsonDocument<192> req;
    if (deserializeJson(req, server.arg("plain")) != DeserializationError::Ok) {
      server.send(400, "text/plain", "JSON error"); return;
    }
    RTCDateTime dt;
    dt.year = req["year"]; dt.month = req["month"]; dt.day = req["day"];
    dt.hour = req["hour"]; dt.minute = req["minute"]; dt.second = req["second"];
    rtc_write_time(dt);
    server.send(200, "application/json", "{\"status\":\"ok\"}");
  });

  server.on("/api/pair", HTTP_POST, []() {
    g_pairing_mode = true;
    server.send(200, "application/json", "{\"status\":\"pairing_active\"}");
  });

  server.serveStatic("/", LittleFS, "/www/index.html");
}

void setup() {
  pinMode(PIN_AMP_SHDN, OUTPUT);
  digitalWrite(PIN_AMP_SHDN, LOW);

  pinMode(PIN_RF_DATA, INPUT);
  attachInterrupt(digitalPinToInterrupt(PIN_RF_DATA), isr_rf_edge, CHANGE);

  Wire.begin(PIN_I2C_SDA, PIN_I2C_SCL);
  LittleFS.begin();
  load_configuration();

  WiFi.mode(WIFI_AP_STA);
  WiFi.softAP("Apartment-Bell", "localonly");
  setup_http_routes();
  server.begin();
}

void loop() {
  server.handleClient();

  if (g_rf_packet_ready) {
    uint32_t packet = g_last_rf_code;
    g_rf_packet_ready = false;
    uint32_t sender_id = packet >> 4;

    if (g_pairing_mode) {
      config.paired_rf_id = sender_id;
      save_configuration();
      g_pairing_mode = false;
      play_audio_file("/audio/intro.raw");
    } else if (sender_id == config.paired_rf_id) {
      play_audio_file(config.chime_file);
    }
  }

  static int last_hour = -1;
  static uint32_t last_check = 0;
  if (millis() - last_check >= 5000) {
    last_check = millis();
    RTCDateTime now = rtc_read_time();
    if (now.valid && now.minute == 0 && now.hour != last_hour) {
      last_hour = now.hour;
      if (config.clock_enabled && !in_quiet_window(now.hour, config.quiet_start, config.quiet_end)) {
        speak_time(now.hour);
      }
    }
  }
}
